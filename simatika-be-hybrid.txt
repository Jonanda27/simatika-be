const { User } = require('../../models');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

exports.login = async (req, res, next) => {
  try {
    const { username, password } = req.body;

    if (!username || !password) {
      return res.status(400).json({ success: false, message: 'Username and password are required' });
    }

    // Cari user berdasarkan username
    const user = await User.findOne({ where: { username } });
    if (!user) {
      return res.status(401).json({ success: false, message: 'Invalid username or password' });
    }

    // Cek password
    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) {
      return res.status(401).json({ success: false, message: 'Invalid username or password' });
    }

    // Buat JWT payload
    const payload = {
      id: user.id,
      username: user.username,
      role: user.role
    };

    // Tanda tangani token (Gunakan JWT_SECRET dari .env atau fallback)
    const secret = process.env.JWT_SECRET || 'simatika_super_secret_key_123';
    const token = jwt.sign(payload, secret, { expiresIn: '7d' });

    res.status(200).json({
      success: true,
      message: 'Login successful',
      token,
      user: payload
    });

  } catch (error) {
    error.message = 'Internal server error: ' + error.message;
    next(error);
  }
};

exports.getMe = async (req, res, next) => {
  try {
    // req.user diset oleh authMiddleware

const db = require('../../models');
const { Umat, Family, Sacrament } = db;

exports.getStats = async (req, res, next) => {
  try {
    // 1. Total Umat
    const totalUmat = await Umat.count();

    // 2. Total Keluarga
    const totalKeluarga = await Family.count();

    // 3. Sakramen Baptis
    const sakramenBaptis = await Sacrament.count({
      where: { jenis_sakramen: 'Baptis' }
    });

    // 4. Sakramen Perkawinan
    const sakramenNikah = await Sacrament.count({
      where: { jenis_sakramen: 'Perkawinan' }
    });

    // 5. Kalkulasi Komposisi Umur (Anak: 0-17, Dewasa: 18-59, Lansia: 60+)
    const umats = await Umat.findAll({
      attributes: ['tanggal_lahir', 'createdAt']
    });

    let anak = 0;
    let dewasa = 0;
    let lansia = 0;

    const currentYear = new Date().getFullYear();
    const currentMonthIndex = new Date().getMonth(); // 0-based

    const umatPerMonth = new Array(12).fill(0);
    let initialTotal = 0;

    umats.forEach((umat) => {
      // Hitung komposisi umur
      if (umat.tanggal_lahir) {
        const birthYear = new Date(umat.tanggal_lahir).getFullYear();
        const age = currentYear - birthYear;

        if (age <= 17) {
          anak++;
        } else if (age >= 18 && age <= 59) {
          dewasa++;
        } else {
          lansia++;
        }
      }

      // Hitung data per bulan
      if (umat.createdAt) {
        const createdDate = new Date(umat.createdAt);
        if (createdDate.getFullYear() < currentYear) {
          initialTotal++;
        } else if (createdDate.getFullYear() === currentYear) {
          umatPerMonth[createdDate.getMonth()]++;
        }
      }
    });

    const komposisiUmur = [
      { name: 'Anak (0-17 th)', value: anak, color: '#3b82f6' },
      { name: 'Dewasa (18-59 th)', value: dewasa, color: '#10b981' },
      { name: 'Lansia (60+ th)', value: lansia, color: '#f59e0b' }
    ];

    // 6. Grafik Umat — Akumulasi pertumbuhan data aktual
    const months = ["Jan", "Feb", "Mar", "Apr", "Mei", "Jun", "Jul", "Agu", "Sep", "Okt", "Nov", "Des"];
    const grafikUmat = [];
    let cumulative = initialTotal;
    
    // Tampilkan hingga bulan saat ini
    for (let i = 0; i <= currentMonthIndex; i++) {
      cumulative += umatPerMonth[i];
      grafikUmat.push({
        name: months[i],
        umat: cumulative
      });
    }

    res.status(200).json({
      success: true,
      data: {
        kpi: {
          totalUmat,
          totalKeluarga,
          sakramenBaptis,
          sakramenNikah
        },
        komposisiUmur,
        grafikUmat
      }
    });

  } catch (error) {
    error.message = 'Gagal memuat statistik dashboard: ' + error.message;
    next(error);
  }
};

const { VerificationQueue } = require('../../models');
const cloudinary = require('cloudinary').v2;

// Konfigurasi Cloudinary dari environment variables
cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET
});

// Helper untuk upload buffer ke Cloudinary
const uploadToCloudinary = (fileBuffer, options = {}) => {
  return new Promise((resolve, reject) => {
    const uploadOptions = {
      folder: options.folder || 'simatika_rumah',
      resource_type: options.resource_type || 'auto'
    };
    
    const uploadStream = cloudinary.uploader.upload_stream(
      uploadOptions,
      (error, result) => {
        if (error) reject(error);
        else resolve(result);
      }
    );
    uploadStream.end(fileBuffer);
  });
};

const processFilesToPayload = async (reqFiles, payload) => {
  if (!reqFiles || reqFiles.length === 0) return payload;

  for (const file of reqFiles) {
    if (file.fieldname === 'foto_rumah') {
      const result = await uploadToCloudinary(file.buffer);
      payload.foto_rumah_url = result.secure_url;
      delete payload.foto_rumah_path;
    } else if (file.fieldname.startsWith('sertifikat_')) {
      // Format: sertifikat_{jenis}_{memberIndex}
      // Contoh: sertifikat_baptis_0
      const parts = file.fieldname.split('_');
      if (parts.length >= 3) {
        const jenis = parts[1];
        const memberIndex = parseInt(parts[2]);
        
        if (!isNaN(memberIndex) && payload.members && payload.members[memberIndex]) {
          const result = await uploadToCloudinary(file.buffer, { 
            folder: 'simatika_sertifikat',
            resource_type: 'auto' // Penting untuk PDF
          });
          
          // Simpan URL ke properti anggota
          payload.members[memberIndex][`${jenis}_sertifikat_url`] = result.secure_url;
        }
      }
    }
  }
  return payload;
};

exports.syncFamilyData = async (req, res, next) => {
  try {
    // 1. Ambil payload string dari request body
    let payload = req.body.payload;
    if (typeof payload === 'string') {
      payload = JSON.parse(payload);
    } else if (!payload && req.body) {
      // Fallback in case it's sent as pure JSON without multipart
      payload = req.body;
    }

    // 2. Upload semua file yang dikirim (foto rumah & sertifikat)
    payload = await processFilesToPayload(req.files, payload);

    // 3. Simpan ke VerificationQueue dengan status PENDING
    const queue = await VerificationQueue.create({
      submitted_by: req.user ? req.user.id : null,
      payload_data: payload,
      status: 'PENDING'
    });

    res.status(201).json({ 
      success: true, 
      message: 'Data berhasil dikirim dan sedang menunggu verifikasi Stasi.', 
      queueId: queue.id 
    });
    
  } catch (error) {
    error.message = 'Gagal mengirim data sinkronisasi: ' + error.message;
    next(error);
  }
};

exports.updateFamilyData = async (req, res, next) => {
  try {
    const queueId = req.params.id;

    // 1. Ambil payload
    let payload = req.body.payload;
    if (typeof payload === 'string') {
      payload = JSON.parse(payload);
    } else if (!payload && req.body) {
      payload = req.body;
    }
    
    // 2. Upload semua file yang dikirim
    payload = await processFilesToPayload(req.files, payload);

    // 3. Cari antrean berdasarkan ID dan pembuat
    const queue = await VerificationQueue.findOne({
      where: { 
        id: queueId,
        submitted_by: req.user.id
      }
    });

    if (!queue) {
      return res.status(404).json({ success: false, message: 'Data tidak ditemukan atau Anda tidak berhak mengubahnya.' });
    }

    // Update data dan reset status
    queue.payload_data = payload;
    queue.status = 'PENDING';
    queue.reject_reason = null;
    await queue.save();

    res.status(200).json({ 
      success: true, 
      message: 'Data berhasil diperbarui dan dikirim ulang untuk verifikasi.', 
      queueId: queue.id 
    });
    
  } catch (error) {
    error.message = 'Gagal memperbarui data: ' + error.message;
    next(error);
  }
};

    if (!user) {
      return res.status(404).json({ success: false, message: 'User not found' });
    }

    // Jika update username, cek apakah sudah dipakai
    if (username && username !== user.username) {
      const existingUser = await User.findOne({ where: { username } });
      if (existingUser) {
        return res.status(400).json({ success: false, message: 'Username already taken' });
      }
      user.username = username;
    }

    if (role) {
      user.role = role;
    }

    // Jika ganti password
    if (password) {
      const salt = await bcrypt.genSalt(10);
      user.password = await bcrypt.hash(password, salt);
    }

    await user.save();

    const userResponse = user.toJSON();
    delete userResponse.password;

    res.status(200).json({ success: true, message: 'User updated successfully', data: userResponse });
  } catch (error) {
    console.error('Error in updateUser:', error);
    res.status(500).json({ success: false, message: 'Internal server error', error: error.message });

// POST /api/verification/:id/approve-paroki
exports.approveParoki = async (req, res) => {
  const transaction = await sequelize.transaction();

  try {
    const queue = await VerificationQueue.findByPk(req.params.id);
    if (!queue) {
      await transaction.rollback();
      return res.status(404).json({ success: false, message: 'Data tidak ditemukan.' });
    }

    if (queue.status !== 'VERIFIED_STASI') {
      await transaction.rollback();
      return res.status(400).json({ success: false, message: 'Data harus diverifikasi oleh Stasi terlebih dahulu.' });
    }

    const data = queue.payload_data;

    // --- LOGIKA INSERT ASLI ---
    let mappedStatus = data.status_rumah || null;
    if (mappedStatus === 'Milik Sendiri') mappedStatus = 'Milik';
    else if (mappedStatus === 'Kontrak/Sewa') mappedStatus = 'Kontrak';

    // 1. Create Family
    const familyData = {
      no_kk_paroki: data.no_kk_paroki || null,
      status_rumah: mappedStatus,
      fasilitas_kamar_mandi: data.has_kamar_mandi === true,
      fasilitas_kamar_cuci: data.has_kamar_cuci === true,
      fasilitas_wc: data.has_wc === true,
      pendapatan_per_bulan: data.pendapatan ? parseInt(data.pendapatan.replace(/\D/g, '')) || null : null,
      asuransi_kesehatan: data.asuransi || null,
      latitude: data.lat || null,
      longitude: data.lng || null
    };

    const newFamily = await Family.create(familyData, { transaction });

    // 2. Loop Members
    const members = data.members || [];
    for (const member of members) {
      const tglLahir = parseDate(member.tgl_lahir);
      let jk = null;
      if (member.jenis_kelamin === 'Laki-Laki') jk = 'L';
      if (member.jenis_kelamin === 'Perempuan') jk = 'P';

      const umatData = {
        family_id: newFamily.id,
        nama_lengkap: member.nama_lengkap || null,
        fam: member.fam || null,
        tempat_lahir: member.tempat_lahir || null,
        tanggal_lahir: tglLahir,
        jenis_kelamin: jk,
        status_dalam_keluarga: member.status_dalam_keluarga || null,
        pekerjaan: member.pekerjaan || null,
        suku: member.suku || null,
        pendidikan_terakhir: member.pendidikan_terakhir || null,
        status_kesehatan: member.status_kesehatan || 'Normal',
        kode_numpang: member.kode_numpang || null
      };

      const newUmat = await Umat.create(umatData, { transaction });

      // 3. Sacraments
      if (member.has_baptis) {
        await Sacrament.create({
          umat_id: newUmat.id,
          jenis_sakramen: 'Baptis',
          nomor_surat: member.baptis_no_surat || null,
          tempat: member.baptis_tempat || null,
          tanggal: parseDate(member.baptis_tgl),
          nama_pastor: member.baptis_pastor || null,
          nama_wali: member.baptis_wali || null
        }, { transaction });
      }
      if (member.has_komuni) {
        await Sacrament.create({
          umat_id: newUmat.id,
          jenis_sakramen: 'Komuni Pertama',
          tempat: member.komuni_tempat || null,
          tanggal: parseDate(member.komuni_tgl),
        }, { transaction });
      }
      if (member.has_krisma) {
        await Sacrament.create({
          umat_id: newUmat.id,
          jenis_sakramen: 'Krisma',
          nomor_surat: member.krisma_no_surat || null,
          tempat: member.krisma_tempat || null,
          tanggal: parseDate(member.krisma_tgl),
          nama_pastor: member.krisma_pastor || null,
          nama_wali: member.krisma_wali || null
        }, { transaction });
      }
      if (member.has_perkawinan) {
        await Sacrament.create({
          umat_id: newUmat.id,
          jenis_sakramen: 'Perkawinan',
          nomor_surat: member.kawin_no_surat || null,
          tempat: member.kawin_tempat || null,
          tanggal: parseDate(member.kawin_tgl),
          nama_pastor: member.kawin_pastor || null,
        }, { transaction });
      }

      // 4. Church Activity
      if (member.misa_frekuensi || member.keaktifan || member.petugas_liturgi) {
        await ChurchActivity.create({
          umat_id: newUmat.id,
          frekuensi_misa: member.misa_frekuensi || null,
          keaktifan: member.keaktifan || null,
          petugas_liturgi: member.petugas_liturgi || null,
          riwayat_kursus: member.kursus_pastoral || null
        }, { transaction });
      }

      // 5. Mutation (Kematian/Pindah)
      if (member.status_hidup === 'Meninggal Dunia') {
        await Mutation.create({
          umat_id: newUmat.id,
          jenis_mutasi: 'Meninggal',
          tujuan: null,
          alasan: member.meninggal_tempat || null,
          tanggal_mutasi: parseDate(member.meninggal_tgl)
        }, { transaction });
      } else if (member.status_pindah && member.status_pindah !== 'Menetap') {
         await Mutation.create({
          umat_id: newUmat.id,
          jenis_mutasi: member.status_pindah,
          tujuan: null,
          alasan: null,
          tanggal_mutasi: new Date()
        }, { transaction });
      }
    }

    // --- END LOGIKA INSERT ASLI ---

    // Update status queue
    queue.status = 'APPROVED_PAROKI';
    await queue.save({ transaction });

    await transaction.commit();
    res.status(200).json({ success: true, message: 'Data disetujui Paroki dan berhasil masuk ke Database.' });
    
  } catch (error) {
    await transaction.rollback();
    console.error('Error in approveParoki:', error);
    res.status(500).json({ success: false, message: 'Gagal memproses persetujuan.', error: error.message });
  }
};

const jwt = require('jsonwebtoken');

const verifyToken = (req, res, next) => {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];

  if (!token) {
    return res.status(401).json({ success: false, message: 'Akses ditolak. Token tidak ditemukan.' });
  }

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET || 'simatika_super_secret_key_123');
    req.user = decoded; // Menyimpan data user (id, username, role) ke request
    next();
  } catch (error) {
    return res.status(403).json({ success: false, message: 'Token tidak valid atau sudah kedaluwarsa.' });
  }
};

module.exports = {
  verifyToken,
};
